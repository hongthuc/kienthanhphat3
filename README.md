    <script>
        window.onload = function () {
            // 1. Menu Điện Thoại
            const mobileMenuBtn = document.getElementById('mobileMenuBtn');
            const mobileMenuDropdown = document.getElementById('mobileMenuDropdown');
            if (mobileMenuBtn && mobileMenuDropdown) {
                mobileMenuBtn.addEventListener('click', function () {
                    mobileMenuDropdown.classList.toggle('hidden');
                });
                document.querySelectorAll('.mobile-nav-link').forEach(function (link) {
                    link.addEventListener('click', function () {
                        mobileMenuDropdown.classList.add('hidden');
                    });
                });
            }

            // 2. Banner 2 Slide Tự Trượt Chuyên Nghiệp
            const slides = document.querySelectorAll('.banner-slide');
            const dots = document.querySelectorAll('.banner-dot');
            const prevBanner = document.getElementById('prevBanner');
            const nextBanner = document.getElementById('nextBanner');
            let currentSlide = 0;
            let slideTimer = null;

            function showSlide(index) {
                if (!slides.length) return;
                currentSlide = (index + slides.length) % slides.length;
                slides.forEach(function (slide, idx) {
                    if (idx === currentSlide) {
                        slide.classList.remove('opacity-0', 'z-0');
                        slide.classList.add('opacity-100', 'z-10');
                    } else {
                        slide.classList.remove('opacity-100', 'z-10');
                        slide.classList.add('opacity-0', 'z-0');
                    }
                });
                dots.forEach(function (dot, idx) {
                    if (idx === currentSlide) {
                        dot.classList.remove('bg-white/50');
                        dot.classList.add('bg-amber-400');
                    } else {
                        dot.classList.remove('bg-amber-400');
                        dot.classList.add('bg-white/50');
                    }
                });
            }

            function startAutoSlide() {
                if (slideTimer) clearInterval(slideTimer);
                slideTimer = setInterval(function () {
                    showSlide(currentSlide + 1);
                }, 4500);
            }

            if (prevBanner) {
                prevBanner.addEventListener('click', function () {
                    showSlide(currentSlide - 1);
                    startAutoSlide();
                });
            }
            if (nextBanner) {
                nextBanner.addEventListener('click', function () {
                    showSlide(currentSlide + 1);
                    startAutoSlide();
                });
            }
            dots.forEach(function (dot) {
                dot.addEventListener('click', function () {
                    const target = parseInt(this.getAttribute('data-slide'), 10);
                    showSlide(target);
                    startAutoSlide();
                });
            });
            startAutoSlide();

            // 3. Lọc Danh Mục Công Trình Thực Tế (9 ảnh)
            const filterBtns = document.querySelectorAll('.construction-filter-btn');
            const constructionItems = document.querySelectorAll('.construction-item');

            function filterConstruction(category) {
                filterBtns.forEach(function (btn) {
                    if (btn.getAttribute('data-cat') === category) {
                        btn.classList.remove('bg-slate-100', 'text-slate-700');
                        btn.classList.add('bg-ktpBlue', 'text-white');
                    } else {
                        btn.classList.remove('bg-ktpBlue', 'text-white');
                        btn.classList.add('bg-slate-100', 'text-slate-700');
                    }
                });

                constructionItems.forEach(function (item) {
                    const itemCat = item.getAttribute('data-category');
                    if (category === 'all' || itemCat === category) {
                        item.classList.remove('hidden');
                    } else {
                        item.classList.add('hidden');
                    }
                });
            }

            filterBtns.forEach(function (btn) {
                btn.addEventListener('click', function () {
                    filterConstruction(this.getAttribute('data-cat'));
                });
            });

            // Nhấn từ thanh danh mục nhanh phía trên
            document.querySelectorAll('[data-filter-jump]').forEach(function (jumpLink) {
                jumpLink.addEventListener('click', function () {
                    const targetCat = this.getAttribute('data-filter-jump');
                    filterConstruction(targetCat);
                });
            });

            // 4. Lọc Danh Mục Dự Án Tại Đà Nẵng (10 ảnh)
            const dnFilterBtns = document.querySelectorAll('.danang-filter-btn');
            const dnItems = document.querySelectorAll('.danang-item');

            dnFilterBtns.forEach(function (btn) {
                btn.addEventListener('click', function () {
                    const selectedCat = this.getAttribute('data-dn-cat');
                    dnFilterBtns.forEach(function (b) {
                        if (b.getAttribute('data-dn-cat') === selectedCat) {
                            b.classList.remove('bg-white', 'text-slate-700');
                            b.classList.add('bg-ktpBlue', 'text-white');
                        } else {
                            b.classList.remove('bg-ktpBlue', 'text-white');
                            b.classList.add('bg-white', 'text-slate-700');
                        }
                    });

                    dnItems.forEach(function (item) {
                        const cat = item.getAttribute('data-dn-category');
                        if (selectedCat === 'all' || cat === selectedCat) {
                            item.classList.remove('hidden');
                        } else {
                            item.classList.add('hidden');
                        }
                    });
                });
            });

            // 5. Xử lý gửi Form Yêu Cầu Báo Giá trên trang chủ
            const quoteForm = document.getElementById('quoteRequestForm');
            const quoteSuccessBox = document.getElementById('quoteSuccessBox');

            if (quoteForm && quoteSuccessBox) {
                quoteForm.addEventListener('submit', function (e) {
                    e.preventDefault();
                    const name = document.getElementById('custName').value.trim();
                    const phone = document.getElementById('custPhone').value.trim();
                    const type = document.getElementById('projectType').value;
                    const budget = document.getElementById('budgetLevel').value;

                    quoteSuccessBox.innerHTML =
                        '<strong>Cảm ơn Quý khách ' + name + '!</strong> Yêu cầu tư vấn & báo giá hạng mục <strong>' +
                        type + '</strong> (Mức đầu tư: ' + budget + ', SĐT: ' + phone +
                        ') đã được ghi nhận. Bộ phận Kỹ thuật <strong>KIẾN THÀNH PHÁT (097927.974)</strong> sẽ liên hệ ngay với Quý khách!';
                    quoteSuccessBox.classList.remove('hidden');
                    quoteForm.reset();
                });
            }
        };
